vim /var/lib/pkgtools/setup/setup.01.mkinitrd 

-------------------------------------------------------------------------------
if [ -z "$INSIDE_INSTALLER" ]; then
  # Don't set $KERNEL... if we have no KERNEL or $1, use the mewest kernel.
  #KERNEL=${KERNEL:-/boot/vmlinuz-generic}
  GENINITRD_NAMED_SYMLINK=${GENINITRD_NAMED_SYMLINK:-false}
  GENINITRD_INITRD_GZ_SYMLINK=${GENINITRD_INITRD_GZ_SYMLINK:-false}
  GENERATOR=${GENERATOR:-mkinitrd}
  DRACUT_OPTS=${DRACUT_OPTS:-"--force --hostonly"}
  AUTOGENERATE_INITRD=${AUTOGENERATE_INITRD:-true}
  AUTO_REMOVE_ORPHANED_INITRDS=${AUTO_REMOVE_ORPHANED_INITRDS:-false}
  AUTO_REMOVE_INITRD_TREE=${AUTO_REMOVE_INITRD_TREE:-false}
  AUTO_UPDATE_GRUB=${AUTO_UPDATE_GRUB:-true}
  GENINITRD_DIALOG=${GENINITRD_DIALOG:-false}
  GENINITRD_COMMAND_OUTPUT=${GENINITRD_COMMAND_OUTPUT:-true}
else # these are the defaults when called from the installer:
  KERNEL=/boot/vmlinuz-generic
  GENINITRD_NAMED_SYMLINK=false
  GENINITRD_INITRD_GZ_SYMLINK=false
  GENERATOR=mkinitrd
  DRACUT_OPTS="--force --hostonly"
  AUTOGENERATE_INITRD=true
  AUTO_REMOVE_ORPHANED_INITRDS=false
  AUTO_REMOVE_INITRD_TREE=false
  AUTO_UPDATE_GRUB=true
  GENINITRD_DIALOG=true
  GENINITRD_COMMAND_OUTPUT=false
fi
-------------------------------------------------------------------------------

- ajuste para gerat initrd compativel

vim /etc/mkinitrd.conf

--------------------------------------------------------------------------------
# antes de rodar o mkinitrd
# rode
# export KERNEL=6.18.34
# para $KERNEL

SOURCE_TREE="/boot/initrd-tree"
CLEAR_TREE="0"
OUTPUT_IMAGE="/boot/initrd-$KERNEL.img"
MODULE_LIST="zfs:xhci-pci:ohci-pci:ehci-pci:xhci-hcd:uhci-hcd:ehci-hcd:hid:usbhid:i2c-hid:hid_generic:hid-asus:hid-cherry:hid-logitech:hid-logitech-dj:hid-logitech-hidpp:hid-lenovo:hid-microsoft:hid_multitouch:xfs"

ROOTFS="zfs"

--------------------------------------------------------------------------------

aqui uso o $KERNEL para não precisar ficar editando o arquivo toda vez

export KERNEL=6.18.34
depmod -a $KERNEL
mkinitrd -k $KERNEL -F /etc/mkinitrd.conf



cd /boot/

export ZPOOL_VDEV_NAME_PATH=YES

mount -t efivarfs none /sys/firmware/efi/efivars

grub-install --target=x86_64-efi --efi-directory=/boot/efi --bootloader-id=slackZFS

grub-mkconfig -o /boot/grub/grub.cfg


após o reboot

criar e adicionar swap

zfs create -V 8G -b $(getconf PAGESIZE) -o compression=lz4 -o logbias=throughput -o sync=always -o primarycache=metadata -o secondarycache=none -o com.sun:auto-snapshot=false zroot/swap

mkswap -f /dev/zvol/zroot/swap

vim /etc/fstab

/dev/zvol/zroot/swap none swap discard 0 0
